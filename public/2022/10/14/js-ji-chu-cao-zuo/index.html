1、将中国标准时间转为 yyyymmdd

export function getYYYYMMDD(string) {
	const d = new Date(string);
	const datetime =
		d.getFullYear() +
		"-" +
		(d.getMonth() + 1) +
		"-" +
		d.getDate() +
		" " +
		d.getHours() +
		":" +
		d.getMinutes() +
		":" +
		d.getSeconds();

	return datetime;
}

3、解析 URL

先根据 ？ 或者 & 符进行将url分割成数组，再遍历数据，根据数组内每项的 = 号，分别取值赋给 key 和 value。最后将 key 和 value 组合成一个对象

// 解析URL
let url = window.location.href;
let reg = /[?&][^?&]+=[^?&]+/g;
let arr = url.match(reg);
let params = {};
if (arr) {
	arr.forEach((item) => {
		let tempArr = item.substring(1).split("=");
		let key = tempArr[0];
		let val = tempArr[1];
		params[key] = val;
	});
}

总体来说就是判断对象里面有多少key，如果key的个数为0，则为空

let isEmpty = false;

//es6
if (Object.keys(object).length === 0) {
	isEmpty = true;
} else {
	isEmpty = false;
}
console.log(isEmpty);

//es5
let arr = [];
for (let i in object) {
	arr.push(i);
}
arr.length === 0 ? (isEmpty = true) : (isEmpty = false);
console.log(isEmpty);

if (JSON.stringify(object) === "{}") {
	isEmpty = true;
} else {
	isEmpty = false;
}
