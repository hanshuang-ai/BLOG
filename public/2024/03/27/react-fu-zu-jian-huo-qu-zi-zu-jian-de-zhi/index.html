    return(
        <div>
            {arr.map(item=>(
                <div key={item}>{item}</div>
            ))}
        </div>
    )
}
// 父组件  负责传值和事件控制
import React, { useState } from 'react';
import Card from "../compoments/card";

function Home() {
    const [ARR, setARR] = useState([1, 2, 3]);
    const handleArr = () => {
        setARR(prevARR => [...prevARR].reverse());
    };
    return (
        <div>
            <Card arr={ARR} />
            <button onClick={handleArr}>翻转数组</button>
        </div>
    );
}
export default Home;
