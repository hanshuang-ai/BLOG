```javascript
function a() {
	// 获取屏幕宽度
	// 屏幕宽度卡片取余
	// 取余后看一行能放下几个卡片
	// 将余数平均分配到每个卡片
	// 如果余数平分后不会造成拥挤到下一行
	// 余数平分后拥挤到下一行
	// 给卡片设置最后宽度
	let screenWidth = window.innerWidth;
	//console.log(screenWidth, "screenWidth");
	let yushu = screenWidth % (100 + 10);
	//console.log(yushu, "yushu");
	let geshu = (screenWidth - yushu) / (100 + 10);
	//console.log(geshu, "geshu");
	let pingjun = ((yushu + 10) / geshu).toFixed(10);
	//console.log(pingjun, "pingjun");
	let dom = document.querySelectorAll(".scroll-list-card");
	//console.log(dom, "dom");
	dom.forEach((item) => {
		// //console.log(item);
		item.style.width = 100 + parseFloat(pingjun) + "px";
	});
}
this.$nextTick(() => {
	a();
	window.addEventListener("resize", () => {
    console.log("resize");
		// a();
	});
});
```
