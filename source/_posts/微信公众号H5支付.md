### 登录授权

> wx.config 在回跳地址时不会输出自己携带的参数，但是确实是传输过去了，可以使用 alert 打印

```javascript
// 准备好了页面回跳地址
let url = "XXXX";

// 请求业务网址，成功后，调用微信授权
axios
  .post(
    "https://tapi.levacp.com/sjsvoice/h5/sys/wechat/jssdk/config/query",
    qs.stringify({ url: url })
  )
  .then((res) => {
    if (res.data.code == 0) {
      // 需要注意的是对回跳地址进行encodeURIComponent编码
      // 另外，如果需要在不同的页面之间传参，可以在传输给后台之前先拼接好参数
      let url = encodeURIComponent(res.data.data.jsSdkConfig.url);
      //配置调试信息
      wx.config({
        debug: false, // 开启调试模式,调用的所有api的返回值会在客户端alert出来，若要查看传入的参数，可以在pc端打开，参数信息会通过log打出，仅在pc端时才会打印。
        appId: "xxx", // 必填，公众号的唯一标识
        timestamp: res.data.data.jsSdkConfig.timestamp, // 必填，生成签名的时间戳
        nonceStr: res.data.data.jsSdkConfig.nonceStr, // 必填，生成签名的随机串
        signature: res.data.data.jsSdkConfig.signature, // 必填，签名
        jsApiList: ["chooseWXPay"], // 必填，需要使用的JS接口列表
      });

      window.location.href = `https://open.weixin.qq.com/connect/oauth2/authorize?appid=wx2d5c3d16592f8f3f&redirect_uri=${url}&response_type=code&scope=snsapi_userinfo&state=STATE#wechat_redirect`;
    } else {
      //普通网页使用vue，并使用vant
      this.$toast(res.data.msg);
    }
  })
  .catch((err) => {
    console.log(err);
  });
```

### 调用微信支付

> 微信支付有两个 API，本文使用的是 WeixinJSBridge

```javascript
let url = window.location.href;
let reg = /[?&][^?&]+=[^?&]+/g;
let arr = url.match(reg);
// 解析URL，将回跳地址带来的参数进行解析
let params = {};
if (arr) {
  arr.forEach((item) => {
    let tempArr = item.substring(1).split("=");
    let key = tempArr[0];
    let val = tempArr[1];
    params[key] = val;
  });
}
//根据code获取用户openid
axios
  .post("URL", qs.stringify({ code: params.code })
  .then((res) => {
    console.log(res, "获取用户Token");
    if (res.data.code == 0) {
      const { accessToken } = res.data.data;
      this.userId = params.userID;
      this.getUID();
      if (params.goodsId == 1) {
        this.value = params.number / 10;
      }

      //    支付
      // const obj = { uid: that.userId, goodsId: this.goodsId, openid: accessToken.openid, quantity: this.quantity }
      const obj = {
        uid: params.userID,
        goodsId: params.goodsId,
        openid: accessToken.openid,
        quantity: params.number,
      };

      //传输openid，请求支付接口
      axios
        .post(
          that.baseURL + "/sjsvoice/h5/sys/create/wx/pay",
          qs.stringify(obj),
          {
            headers: {
              channel: "wxH5",
              app_type: "4",
            },
          }
        )
        .then((res) => {
          console.log(res, "支付结果");
          console.log(res.data.data);

          // 微信支付API
          function onBridgeReady() {
            let { package } = res.data.data;
            WeixinJSBridge.invoke(
              "getBrandWCPayRequest",
              {
                appId: "appid", //公众号ID，由商户传入
                timeStamp: res.data.data.timeStamp, //时间戳，自1970年以来的秒数
                nonceStr: res.data.data.nonceStr, //随机串
                package: "prepay_id=" + res.data.data.package, //一定要注意这个形式、一定要注意这个形式、一定要注意这个形式
                signType: "RSA", //微信签名方式：
                paySign: res.data.data.paySign, //微信签名
              },
              function (res) {
                if (res.err_msg == "get_brand_wcpay_request:ok") {
                  // 使用以上方式判断前端返回,微信团队郑重提示：
                  //res.err_msg将在用户支付成功后返回ok，但并不保证它绝对可靠。
                }
              }
            );
          }
          if (typeof WeixinJSBridge == "undefined") {
            if (document.addEventListener) {
              document.addEventListener(
                "WeixinJSBridgeReady",
                onBridgeReady,
                false
              );
            } else if (document.attachEvent) {
              document.attachEvent("WeixinJSBridgeReady", onBridgeReady);
              document.attachEvent("onWeixinJSBridgeReady", onBridgeReady);
            }
          } else {
            onBridgeReady();
          }
        })
        .catch((err) => {
          console.log(err);
        });
    }
  })
  .catch((err) => {
    console.log(err);
  });
```
