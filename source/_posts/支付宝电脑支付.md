## 封装请求后台的接口

```javascript
// 先封装请求后端的接口(promise)
/**
 * 支付宝支付
 * @param goodsId  1为自定义
 * @param quantity 钻石数量
 */
export function zhifubao(goodsId, quantity) {
  return axios.post(
    //后端支付接口
    store.state.baseURL + "/sjsvoice/c/pms/create/zfb/pay",
    qs.stringify({
      goodsId: goodsId,
      quantity: quantity,
    }),
    {
      headers: {
        "user-id": store.state.userInfo.userId,
        "user-token": store.state.userToken,
        channel: "pc",
        "app-type": "3",
      },
    }
  );
}
```

## 在其他页面使用接口

```javascript
// 在其他页面引入后使用
zhifubao(payTtype, number)
  .then((res) => {
    console.log(res);
    console.log(res.data);
    if (res.data) {
      this.showPayPage = true;
      //提前预留一个dom元素区域来承展form表单
      this.html = res.data;
      this.$nextTick(() => {
        //使用这个，直接打开form表单，然后扫码即可
        document.forms[0].submit();
      });
    }
  })
  .catch((err) => {
    console.log(err);
  });
```
